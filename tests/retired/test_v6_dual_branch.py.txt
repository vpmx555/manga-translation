class MemoryIO:
    def __init__(self, config):
        self.store = SimpleNamespace(manifest={"config": config})
        self.values = {}

    def get(self, key, request):
        return copy.deepcopy(self.values.get((key, fingerprint(request))))

    def put(self, key, request, result):
        self.values[key, fingerprint(request)] = copy.deepcopy(result)


class SemanticFake:
    def __init__(self):
        self.requests = []

    def infer_once(self, task, prompt, data, schema, validator, images=None):
        self.requests.append(copy.deepcopy(data))
        result = {"utterances": {}, "distinct_person_pairs": []}
        for target in data["targets"]:
            allowed = schema["properties"]["utterances"]["properties"][target["id"]]["properties"]["speaker_person_id"]["enum"]
            speaker = allowed[0] if len(allowed) == 1 else "person:b" if data["direction"] == "forward" else "person:a"
            result["utterances"][target["id"]] = {"content_type": "dialogue", "speech_style": "normal",
                "speaker_person_id": speaker, "speaker_cluster_ref": None, "addressee_person_ids": [], "mentions": {},
                "evidence": "continuity fixture", "speaker_kind": "person", "addressee_type": "unknown",
                "inference_relation": "continue", "evidence_refs": [], "alternative_candidates": [], "is_aside": False,
                "speaker_correction_reason": "backward fixture", "speech_style_evidence": "normal voice"}
        validator(result)
        return result


class SceneFake:
    def infer_once(self, task, prompt, data, schema, validator, images=None):
        result = {"decisions": {r["id"]: {"relation": "scene_change" if r["id"] == "d2" else "continue",
            "reason": "semantic fixture", "main_summary": "topic", "evidence_refs": []} for r in data["targets"]}}
        validator(result)
        return result


class DualBranchFixtures(unittest.TestCase):
    def test_routing_independent_of_role_and_style(self):
        rows, raw, state, bank = fixture()
        rows[0].update(content_type="dialogue", speech_style="shout", visual_evidence=[{"bubble_style": "caption_hint"}])
        rows[1].update(content_type="thought", speech_style="thought")
        result = route_and_anchor(rows, raw, state, bank)
        self.assertEqual([r["analysis_branch"] for r in result], ["caption", "dialogue"])
        self.assertEqual([r["essential_index"] for r in result], [1,2])

    def test_anchor_requires_same_source_tail_character_and_stable_bank(self):
        rows, raw, state, bank = fixture()
        result = route_and_anchor(rows, raw, state, bank)
        self.assertTrue(result[0]["speaker_anchor"]["locked"])
        raw["pages"][0]["text_tail_associations"] = [[1,0]]
        self.assertFalse(route_and_anchor(rows, raw, state, bank)[0]["speaker_anchor"]["locked"])
        raw["pages"][0]["text_tail_associations"] = [[0,0]]
        self.assertFalse(route_and_anchor(rows, raw, state, {"characters": {}})[0]["speaker_anchor"]["locked"])

    def test_caption_inclusive_gap_and_chain(self):
        self.assertTrue(caption_link(1,7,7))
        self.assertFalse(caption_link(1,8,7))
        self.assertTrue(caption_link(1,6,7) and caption_link(6,11,7))
        self.assertFalse(caption_link(1,11,7))

    def test_previous_extraction_cannot_supply_current_anchor(self):
        rows, raw, state, bank = fixture()
        state["extraction_scope"] = "current"
        state["observations"] = {"old:c0": state["observations"]["c0"],
                                 "current:c0": state["observations"]["c0"],
                                 "current:c1": state["observations"]["c1"]}
        state["cluster_to_person"] = {"old:c0": "person:b", "current:c0": "person:a", "current:c1": "person:b"}
        result = route_and_anchor(rows, raw, state, bank)
        self.assertEqual(result[0]["speaker_anchor"]["person_id"], "person:a")
        self.assertEqual([p["cluster_ref"] for p in result[0]["original_speaker"]], ["current:c0"])

    def test_anchor_bank_must_match_current_character_detection(self):
        rows, raw, state, bank = fixture()
        state["observations"]["c0"][0]["source_bank_id"] = 2
        result = route_and_anchor(rows, raw, state, bank)
        self.assertFalse(result[0]["speaker_anchor"]["locked"])

    def test_caption_scene_position_does_not_read_caption_content(self):
        config = load_config()
        rows = [{"id": "c0", "analysis_branch": "caption", "is_filler": False},
                {"id": "d1", "analysis_branch": "dialogue", "text": "topic one", "is_filler": False},
                {"id": "fill", "analysis_branch": "dialogue", "text": "uh", "is_filler": True},
                {"id": "c1", "analysis_branch": "caption", "is_filler": False},
                {"id": "d2", "analysis_branch": "dialogue", "text": "topic two", "is_filler": False},
                {"id": "c2", "analysis_branch": "caption", "is_filler": False}]
        scenes = scene_plan(rows, MemoryIO(config), SceneFake(), config)
        self.assertEqual(len(scenes), 2)
        self.assertEqual(rows[0]["scene_id"], rows[1]["scene_id"])
        self.assertEqual(rows[2]["relation"], "filler")
        self.assertEqual(rows[3]["scene_id"], rows[1]["scene_id"])
        self.assertEqual(rows[5]["scene_id"], rows[4]["scene_id"])

    def test_two_passes_keep_anchor_and_use_only_branch_content(self):
        rows, raw, state, bank = fixture()
        rows = route_and_anchor(rows, raw, state, bank)
        for row in rows:
            row.update(scene_id="scene:1", thread_id="main:1", scene_boundary_reason="fixture", caption_max_distance=7)
        rows[1]["speaker_anchor"] = {"locked": False, "person_id": None, "sources": []}
        config, provider = load_config(), SemanticFake()
        participants = {p: [{"source": "fixture", "target_id": f"u{i}", "cluster_ref": f"c{i}"}] for i, p in enumerate(("person:a", "person:b"))}
        result, checkpoints = run_branch("dialogue", rows, {"scene_id": "scene:1", "participants": participants},
            state, {}, MemoryIO(config), provider, config, lambda: None)
        self.assertEqual(result["u0"]["speaker_person_id"], "person:a")
        self.assertEqual(result["u1"]["speaker_person_id"], "person:a")
        self.assertEqual({c["direction"] for c in checkpoints}, {"forward", "backward"})
        self.assertTrue(all(r["analysis_branch"] == "dialogue" for r in provider.requests))
        self.assertTrue(all(result[k]["final_status"] == "provisional" for k in result))
